# 云韵音乐 · 后端一键编译(Windows + Linux)
#
# 用法:整段/逐行拷到 PowerShell 里执行(和 编译.txt / 编译2.txt 同一套用法)
#
# 两个注意点:
#   1. 先停掉正在运行的后端 —— Windows 上被占用的 exe 没法覆盖,会报 Access is denied
#   2. GOROOT 必须是**完整的** Go 安装;有些精简包缺标准库源码,
#      会报 "package xxx is not in std",那种目录编译不了(换完整包即可)
#
# 产物目录:go_backend\
#   yunyun-server.exe    Windows 64 位
#   yunyun-server-linux  Linux  64 位
# 两个都是「静态、免依赖」可执行文件:Linux 上 chmod +x 后直接跑。

# ========== 1) 只编 Windows → go_backend\yunyun-server.exe ==========
$env:GOROOT="D:\go1.22.12.windows-amd64\go"; $env:PATH="$env:GOROOT\bin;$env:PATH"; Set-Location "c:\Users\Administrator\Desktop\yin\go_backend"; go build -o yunyun-server.exe . 2>&1 | Select-Object -Last 20; Get-Item .\yunyun-server.exe | Select-Object Name,@{n='MB';e={[math]::Round($_.Length/1MB,1)}},LastWriteTime

# ========== 2) 只编 Linux → go_backend\yunyun-server-linux ==========
# 注:跑完这个窗口的 GOOS/GOARCH 就变成 linux 了,后面要编 Windows 记得清掉(见下面第 4 条)
$env:GOROOT="D:\go1.22.12.windows-amd64\go"; $env:PATH="$env:GOROOT\bin;$env:PATH"; $env:GOOS="linux"; $env:GOARCH="amd64"; $env:CGO_ENABLED="0"; Set-Location "c:\Users\Administrator\Desktop\yin\go_backend"; go build -o yunyun-server-linux . 2>&1 | Select-Object -Last 20; Get-Item .\yunyun-server-linux | Select-Object Name,@{n='MB';e={[math]::Round($_.Length/1MB,1)}},LastWriteTime; Remove-Item Env:GOOS,Env:GOARCH,Env:CGO_ENABLED -ErrorAction SilentlyContinue

# ========== 3) 一键两个都编(推荐:编完自动清理环境变量)==========
$env:GOROOT="D:\go1.22.12.windows-amd64\go"; $env:PATH="$env:GOROOT\bin;$env:PATH"; $env:CGO_ENABLED="0"; Set-Location "c:\Users\Administrator\Desktop\yin\go_backend"; go build -o yunyun-server.exe . ; $env:GOOS="linux"; $env:GOARCH="amd64"; go build -o yunyun-server-linux . ; Remove-Item Env:GOOS,Env:GOARCH,Env:CGO_ENABLED -ErrorAction SilentlyContinue; Get-ChildItem .\yunyun-server* | Select-Object Name,@{n='MB';e={[math]::Round($_.Length/1MB,1)}},LastWriteTime

# ========== 4) 附:只做编译检查(不出二进制、最快,改完代码先跑这个)==========
$env:GOROOT="D:\go1.22.12.windows-amd64\go"; $env:PATH="$env:GOROOT\bin;$env:PATH"; Set-Location "c:\Users\Administrator\Desktop\yin\go_backend"; go vet ./... ; go build ./... ; "OK=$LASTEXITCODE"
